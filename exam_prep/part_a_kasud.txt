### 1. ПАПКА + DOCKERFILE #############################################

mkdir -p ~/ica0022/exam_run && cd ~/ica0022/exam_run
# app.py + requirements.txt из репо экзамена

cat > Dockerfile <<'EOF'
FROM python:3.12-slim-bookworm AS builder
WORKDIR /build
RUN python -m venv /opt/venv
COPY requirements.txt .
RUN /opt/venv/bin/pip install --no-cache-dir -r requirements.txt

FROM python:3.12-slim-bookworm AS runtime
ENV PATH="/opt/venv/bin:$PATH"
ENV PYTHONDONTWRITEBYTECODE=1 PYTHONUNBUFFERED=1
WORKDIR /app
RUN groupadd --gid 10001 app && useradd --uid 10001 --gid app --no-create-home app
COPY --from=builder /opt/venv /opt/venv
COPY app.py .
USER 10001:10001
EXPOSE 8080
CMD ["waitress-serve", "--listen=0.0.0.0:8080", "app:app"]
EOF

cat > .dockerignore <<'EOF'
__pycache__/
*.pyc
venv/
.venv/
.git
.gitignore
Dockerfile
.dockerignore
chart/
*.md
EOF

    cpu: 100m

  enabled: true
  storageClass: ""

  enabled: true
  scripts:
    init.sql: |
      CREATE TABLE IF NOT EXISTS counter (
          id integer PRIMARY KEY CHECK (id = 1),
          visits bigint NOT NULL DEFAULT 0

      ON CONFLICT (id) DO NOTHING;

### 6. INGRESS (ALB) ##################################################

kubectl create ingress app \
  --class=alb \
  --rule="/*=app:8080" \
  --annotation=alb.ingress.kubernetes.io/scheme=internet-facing \
  --annotation=alb.ingress.kubernetes.io/target-type=ip \
  --annotation=alb.ingress.kubernetes.io/healthcheck-path=/readyz \
  --dry-run=client -o yaml > k8s/app-ingress.yaml

kubectl apply -f k8s/app-ingress.yaml
kubectl get ingress -w
kubectl describe ingress app

curl -i http://<ADDRESS>/readyz
curl -i http://<ADDRESS>/
